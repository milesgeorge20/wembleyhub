    <script>
        async function sendLog() {
            const statusDiv = document.getElementById('logStatus');
            statusDiv.innerText = "Sending log...";
            try {
                const response = await fetch('/api/ledger', {
                    method: 'POST',
                    headers: { 'Content-Type': 'application/json' },
                    body: JSON.stringify({ action: 'Test Session Boot', amount: 5.00, staff_id: 'STAFF_01' })
                });
                const data = await response.json();
                if(data.success) {
                    statusDiv.innerHTML = `<span style="color:#00ff87">✓ Log saved securely to SQLite! Log ID: ${data.logId}</span>`;
                }
            } catch (err) {
                statusDiv.innerHTML = `<span style="color:#ff4757">Error writing log: ${err.message}</span>`;
            }
        }

        function toggleRelay(id, state) {
            document.getElementById('relayStatus').innerText = `Sending relay command to turn ${state}... (Check terminal for activity)`;
        }
    </script>
